cmake_minimum_required(VERSION 3.20)
project(LauncherPanelContract CXX)
include("${YI_PANEL_HELPER}")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/input")
set(original "${CMAKE_CURRENT_BINARY_DIR}/input/launcher_imgui.cpp")
file(WRITE "${original}" "void fixture() {\n    draw_controllers_row(m, *th);\n}\n")
set(index 0)
foreach(order "YiRaDrawPanel;YiEffectsDrawLauncher;YiWsDrawLauncher"
              "YiRaDrawPanel;YiWsDrawLauncher;YiEffectsDrawLauncher"
              "YiEffectsDrawLauncher;YiRaDrawPanel;YiWsDrawLauncher"
              "YiEffectsDrawLauncher;YiWsDrawLauncher;YiRaDrawPanel"
              "YiWsDrawLauncher;YiRaDrawPanel;YiEffectsDrawLauncher"
              "YiWsDrawLauncher;YiEffectsDrawLauncher;YiRaDrawPanel")
  math(EXPR index "${index}+1")
  set(target "fixture_${index}")
  add_library(${target} STATIC "${original}")
  foreach(callback IN LISTS order)
    yi_launcher_panel(${target} ${callback})
  endforeach()
  list(GET order 0 first)
  yi_launcher_panel(${target} ${first})
  set(generated "${CMAKE_CURRENT_BINARY_DIR}/yi-launcher/${target}/launcher_imgui.cpp")
  file(READ "${generated}" text)
  foreach(callback IN LISTS order)
    string(REGEX MATCHALL "::${callback}\\(" calls "${text}")
    list(LENGTH calls count)
    if(NOT count EQUAL 1)
      message(FATAL_ERROR "${target}: ${callback} has ${count} calls")
    endif()
  endforeach()
endforeach()
file(READ "${original}" unchanged)
if(NOT unchanged STREQUAL "void fixture() {\n    draw_controllers_row(m, *th);\n}\n")
  message(FATAL_ERROR "The original launcher source was modified")
endif()
message(STATUS "PASS all six panel registration orders, deduplication, target isolation and original source preservation")
